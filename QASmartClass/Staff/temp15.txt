                else

                    txtWelcome.Text = $"Xin chào, {studentName} 👋";
                }

                var todayLessons = db.Lessons
                    .Where(l => l.ScheduledFor != null && l.ScheduledFor.Value.Date == DateTime.Today)
                    .OrderBy(l => l.ScheduledFor)
                    .ToList();
                int lessonCount = todayLessons.Count;

                // 3. Load pending assignments (Bài tập sắp hết hạn)
                var pendingAssignments = db.Assignments
                    .Where(a => a.Deadline >= DateTime.Now)
                    .OrderBy(a => a.Deadline)
                    .ToList();
                int pendingCount = pendingAssignments.Count;

                // 4. Load recent scores (Điểm số gần nhất)
                var submitCount = db.QuizResults.Count(r => r.StudentId == studentId);
                double avgScore = 0;

                    try { avgScore = db.QuizResults.Where(r => r.StudentId == studentId).Average(r => (double)r.Score) / 10.0; }
                    catch { avgScore = 0; }
                }
