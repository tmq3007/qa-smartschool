        private void SaveHomeworkToDb(HomeworkItem hw, bool isUpdate)
        {
            try
            {
                var app = (QASmartTouch.App)Application.Current;
                var json = System.Text.Json.JsonSerializer.Serialize(hw);

                if (isUpdate && hw.EventLogId > 0)
                {
                    var existingLog = app.Database.EventLogs.Find(hw.EventLogId);
                    if (existingLog != null)
                    {
                        existingLog.Details = json;
                        existingLog.Timestamp = DateTime.Now;
                    }
                }
                else
                {
                    var log = new QASmartClass.Data.EventLog
                    {
                        EventType = "HOMEWORK",
                        Actor = "GV",
                        Details = json,
                        Timestamp = DateTime.Now
                    };
                    app.Database.EventLogs.Add(log);
                }

                app.Database.SaveChanges();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Lỗi lưu: {ex.Message}", "Lỗi", MessageBoxButton.OK, MessageBoxImage.Error);
            }
        }
